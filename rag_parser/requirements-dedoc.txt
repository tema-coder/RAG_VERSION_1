# Runtime dependencies for `DedocParser` (parsers/dedoc_parser.py).
#
# dedoc 2.8 is installed WITHOUT its own dependency metadata:
#
#     pip install --no-deps dedoc==2.8
#     pip install -r rag_parser/requirements-dedoc.txt
#
# Reason: dedoc 2.8 pins starlette<=0.27.0, uvicorn<=0.23.2,
# python-multipart==0.0.6 and python-docx==0.8.11 (incompatible with
# fastapi==0.115.0 used by this service) and additionally pins
# opencv-python<4.6, scipy<=1.11.2, scikit-image<=0.21 and
# scikit-learn<=1.2.2, which have no wheels for Python 3.12 and fail to
# build from source. The pins below are the modern equivalents actually used
# on the DOCX path; `pip check` will keep reporting dedoc's own metadata.
#
# `xlrd<2` is mandatory: dedoc.readers touches `xlrd.xlsx` at class level.
# `numpy<2` keeps streamlit 1.38 and the semantic chunker stack working.
# `torch` (CPU build) is imported by dedoc's docx attachments extractor:
#     pip install torch --index-url https://download.pytorch.org/whl/cpu
numpy<2
pandas<3
scipy
scikit-image
scikit-learn
opencv-python-headless
pymupdf
xgboost
xgbfir
xlrd==1.2.0
roman
rtree
imutils
joblib
Levenshtein
fonttools
piexif
puremagic
pypdf
pypdfium2
pytesseract
olefile
beautifulsoup4
charset-normalizer
python-dateutil
huggingface_hub
ujson
dedoc-utils==0.3.11
